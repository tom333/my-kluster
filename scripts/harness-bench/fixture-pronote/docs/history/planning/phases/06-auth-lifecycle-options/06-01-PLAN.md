    @pytest.mark.parametrize("tz_name", ["Europe/Paris", "Pacific/Noumea"])
    def test_compute_interval_respects_adaptive_disabled(tz_name: str) -> None:
        """Phase 6 OPT-02 / D-09 — adaptive_enabled=False bypasses every adaptive branch.

        At 18:00 Thursday Pacific/Noumea with tomorrow=school-day, adaptive_enabled=True
        would hit the afternoon-window branch (15 min). With adaptive_enabled=False we
        always return refresh_interval (30 min) ± jitter.
        """
        from datetime import date, datetime, time, timedelta
        from zoneinfo import ZoneInfo
        import random

        from custom_components.ha_pronote.politesse import PolitesseOptions, compute_interval

        school_tz = ZoneInfo("Pacific/Noumea")  # decision tree is school-tz-anchored
        options = PolitesseOptions(
            school_tz=school_tz,
            refresh_interval=timedelta(minutes=30),
            afternoon_interval=timedelta(minutes=15),
            afternoon_window=(time(17, 0), time(20, 0)),
            quiet_hours=(time(22, 0), time(6, 0)),
            suspended_cadence=timedelta(hours=6),
            quiet_cadence=timedelta(hours=4),
            vacation_ranges=(),
            holiday_dates=frozenset(),
            jitter_seconds=30,
            adaptive_enabled=False,
        )
        # 2026-05-07 is a Thursday (NC weekday, school day); 18:00 Pacific/Noumea
        # is inside the 17h-20h afternoon-window branch. With adaptive=False we bypass.
        now = datetime(2026, 5, 7, 18, 0, tzinfo=ZoneInfo(tz_name)).astimezone(school_tz)
        rng = random.Random(42)
        interval = compute_interval(now, options, rng=rng)
        # 30 min ± 30s → in [29:30, 30:30]; never 15 min (the bypassed afternoon branch).
        assert timedelta(minutes=29, seconds=30) <= interval <= timedelta(minutes=30, seconds=30)
