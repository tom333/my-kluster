```python
def _grade_from_raw(raw: Any) -> Grade:
    """Field-by-field copy of a pronotepy grade.

    pronotepy attr → Grade model field mapping (RESEARCH.md "Name mapping alert"):
      raw.average → Grade.class_average  (pronotepy uses "average" for class avg)
      raw.max     → Grade.class_max
      raw.min     → Grade.class_min
      raw.comment → Grade.comment
    getattr(..., None) fallback: missing fields return "" (explicit visible default,
    not a swallowing catch — per project feedback: no silent exceptions).
    """
    return Grade(
        subject=raw.subject.name if raw.subject else "",
        value=str(raw.grade) if raw.grade is not None else "",
        out_of=str(raw.out_of) if raw.out_of is not None else "",
        coefficient=str(raw.coefficient) if raw.coefficient is not None else "",
        date=raw.date,
        class_average=str(getattr(raw, "average", None) or ""),    # Phase 4 — pronotepy attr is "average"
        class_min=str(getattr(raw, "min", None) or ""),            # Phase 4 — pronotepy attr is "min"
        class_max=str(getattr(raw, "max", None) or ""),            # Phase 4 — pronotepy attr is "max"
        comment=str(getattr(raw, "comment", None) or ""),          # Phase 4 — empty or None → ""
    )
```
