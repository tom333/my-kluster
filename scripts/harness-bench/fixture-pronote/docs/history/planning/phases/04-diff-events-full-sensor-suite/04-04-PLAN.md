```python
@property
def device_info(self) -> DeviceInfo:
    """D-17 extended with Phase 4 D-19 — DeviceInfo.model = class level from ClientInfo.

    CLASS_LEVEL_ATTR = "class_name" — probe-confirmed in PHASE-4-PROBE-NOTES.md STEP 11.
    getattr(..., None): explicit visible default (not a swallowing catch — no silent exceptions).
    ClientInfo.class_name returns "" when absent, not None — "or None" converts "" to None.
    None hides the "Model" row in HA's device panel (acceptable behaviour per D-19).
    """
    from .const import CLASS_LEVEL_ATTR  # local import avoids any circular at module-init time
    client = self._entry.runtime_data.client
    class_label = getattr(client.info, CLASS_LEVEL_ATTR, None)
    return DeviceInfo(
        identifiers={(DOMAIN, self._entry.runtime_data.child_identifier)},
        name=self._entry.data["child_name"],
        manufacturer="Pronote",
        model=class_label or None,  # D-19: "" → None, None stays None
    )
```
